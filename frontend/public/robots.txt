# /api/* is on a different origin but listed here defensively.

# Block the Piston code-execution endpoint path
Disallow: /api/execute

# ── Static assets: no need to index ──────────────────────────────────────────
Disallow: /assets/
Disallow: /*.chunk.js$
Disallow: /*.map$
